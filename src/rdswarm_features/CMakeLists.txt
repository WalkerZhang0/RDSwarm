cmake_minimum_required(VERSION 3.10)
project(rdswarm_features)
find_package(catkin REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc dnn imgcodecs)
catkin_package(INCLUDE_DIRS include LIBRARIES rdswarm_features DEPENDS OpenCV)
add_library(rdswarm_features src/superpoint_cpu.cpp)
target_compile_features(rdswarm_features PUBLIC cxx_std_17)
target_include_directories(rdswarm_features PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(rdswarm_features PUBLIC ${OpenCV_LIBS})
if(CATKIN_ENABLE_TESTING)
  add_executable(superpoint_sampling_test test/sampling_test.cpp)
  target_link_libraries(superpoint_sampling_test rdswarm_features)
  add_test(NAME rdswarm_superpoint_sampling COMMAND $<TARGET_FILE:superpoint_sampling_test>)
  add_executable(superpoint_model_probe test/model_probe.cpp)
  target_link_libraries(superpoint_model_probe rdswarm_features)
endif()
install(TARGETS rdswarm_features LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
  ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
