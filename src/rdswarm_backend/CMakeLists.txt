cmake_minimum_required(VERSION 3.10)
project(rdswarm_backend)
find_package(catkin REQUIRED COMPONENTS rdswarm_core rdswarm_msgs)
find_package(GTSAM 4.2 REQUIRED)
set(GTSAM_INCLUDE_DIRS ${GTSAM_INCLUDE_DIR})
set(GTSAM_LIBRARIES gtsam)
catkin_package(INCLUDE_DIRS include LIBRARIES rdswarm_backend_contract CATKIN_DEPENDS rdswarm_core rdswarm_msgs DEPENDS GTSAM)
add_library(rdswarm_backend_contract src/backend_contract.cpp)
target_compile_features(rdswarm_backend_contract PUBLIC cxx_std_17)
target_include_directories(rdswarm_backend_contract PUBLIC include ${catkin_INCLUDE_DIRS} ${GTSAM_INCLUDE_DIR})
target_link_libraries(rdswarm_backend_contract PUBLIC gtsam ${catkin_LIBRARIES})
add_dependencies(rdswarm_backend_contract ${catkin_EXPORTED_TARGETS})
if(CATKIN_ENABLE_TESTING)
  add_executable(constraint_contract_test test/constraint_contract_test.cpp)
  target_link_libraries(constraint_contract_test rdswarm_backend_contract)
endif()
install(TARGETS rdswarm_backend_contract LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
  ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
